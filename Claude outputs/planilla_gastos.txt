/**
 * ============================================================
 *  Planilla de gastos - recibe las boletas desde el telefono
 * ============================================================
 *
 * Pega esto en Extensiones > Apps Script de tu planilla, guarda,
 * y publicalo como aplicacion web con acceso "Cualquier persona".
 *
 * Crea solo tres hojas:
 *   Documentos - una fila por boleta
 *   Detalle    - una fila por producto
 *   Resumen    - totales por dia, por mes y por ano
 *
 * No borra nada ni toca otras hojas que tengas.
 */

/* ------------------------------------------------------------------
   TU CLAVE. Cambia el texto de abajo por cualquier cosa larga e
   inventada, sin espacios. Por ejemplo: 'perro-azul-7741-quillota'

   Sirve para esto: la direccion del script queda abierta porque la app
   no tiene cuenta de Google con la cual identificarse. Con clave, aunque
   alguien consiga la direccion, no puede escribir nada sin ella.

   Despues, a la direccion que pegas en la app le agregas al final:
       ?clave=perro-azul-7741-quillota

   Si dejas la clave vacia (''), el script acepta a cualquiera. No es
   lo recomendable, pero sirve para probar.
   ------------------------------------------------------------------ */
var CLAVE = 'gastos-4417-quillota';

var HOJA_DOC = 'Documentos';
var HOJA_DET = 'Detalle';
var HOJA_RES = 'Resumen';

var COL_DOC = ['ID', 'Fecha', 'Mes', 'Ano', 'Tipo', 'N Documento', 'RUT emisor',
               'Neto', 'IVA', 'Total', 'Categoria', 'Tarjeta', 'Forma pago',
               'Avisos', 'Recibido'];
var COL_DET = ['ID', 'Fecha', 'N Documento', 'Descripcion', 'Cantidad',
               'Valor unitario', 'Valor total'];

function doPost(e) {
  // El candado evita que dos envios seguidos escriban en la misma fila.
  var candado = LockService.getScriptLock();
  try {
    candado.waitLock(20000);
  } catch (err) {
    return responder({ ok: false, error: 'ocupado' });
  }
  try {
    // La clave viaja en la direccion, no en el cuerpo, para poder cambiarla
    // sin tocar la app: basta con pegar una direccion nueva.
    if (CLAVE !== '') {
      var recibida = (e && e.parameter && e.parameter.clave) || '';
      if (recibida !== CLAVE) {
        return responder({ ok: false, error: 'clave incorrecta' });
      }
    }

    var datos = JSON.parse(e.postData.contents);

    if (datos.prueba) {
      preparar();
      return responder({ ok: true, mensaje: 'conexion correcta' });
    }

    var libro = preparar();
    var hojaDoc = libro.getSheetByName(HOJA_DOC);

    // Si ese ID ya esta, no se escribe de nuevo. Es lo que permite que el
    // telefono reintente sin miedo: reenviar cien veces deja una sola fila.
    if (existe(hojaDoc, datos.id)) {
      return responder({ ok: true, mensaje: 'ya estaba' });
    }

    var fecha = datos.fecha || '';
    var mes = fecha.length >= 7 ? fecha.substring(0, 7) : '';
    var ano = fecha.length >= 4 ? fecha.substring(0, 4) : '';

    hojaDoc.appendRow([
      String(datos.id),
      fecha,
      mes,
      ano,
      datos.tipo || '',
      datos.numero || '',
      datos.rut || '',
      numero(datos.neto),
      numero(datos.iva),
      numero(datos.total),
      datos.categoria || '',
      datos.tarjeta || '',
      datos.formaPago || '',
      datos.avisos || '',
      new Date(),
    ]);

    var items = datos.items || [];
    if (items.length > 0) {
      var hojaDet = libro.getSheetByName(HOJA_DET);
      var filas = [];
      for (var i = 0; i < items.length; i++) {
        var it = items[i];
        filas.push([
          String(datos.id),
          fecha,
          datos.numero || '',
          it.descripcion || '',
          numero(it.cantidad),
          numero(it.valorUnitario),
          numero(it.valorTotal),
        ]);
      }
      hojaDet.getRange(hojaDet.getLastRow() + 1, 1, filas.length, COL_DET.length)
             .setValues(filas);
    }

    var donde = alCashflow(libro, datos, fecha);

    return responder({ ok: true, mensaje: 'guardado', cashflow: donde });
  } catch (err) {
    return responder({ ok: false, error: String(err) });
  } finally {
    candado.releaseLock();
  }
}

/** Abrir la direccion en el navegador tiene que decir algo util. */
function doGet(e) {
  if (CLAVE !== '') {
    var recibida = (e && e.parameter && e.parameter.clave) || '';
    if (recibida !== CLAVE) {
      return responder({ ok: false, error: 'clave incorrecta' });
    }
  }
  return responder({ ok: true, mensaje: 'La planilla esta escuchando.' });
}

function responder(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/** '' y null quedan como celda vacia; lo demas como numero de verdad. */
function numero(v) {
  if (v === null || v === undefined || v === '') return '';
  var n = Number(v);
  return isNaN(n) ? '' : n;
}

function existe(hoja, id) {
  var ultima = hoja.getLastRow();
  if (ultima < 2) return false;
  var columna = hoja.getRange(2, 1, ultima - 1, 1).getValues();
  var buscado = String(id);
  for (var i = 0; i < columna.length; i++) {
    if (String(columna[i][0]) === buscado) return true;
  }
  return false;
}

/** Crea las hojas que falten. Si ya existen, no las toca. */
function preparar() {
  var libro = SpreadsheetApp.getActiveSpreadsheet();
  crearHoja(libro, HOJA_DOC, COL_DOC);
  crearHoja(libro, HOJA_DET, COL_DET);
  crearResumen(libro);
  return libro;
}

function crearHoja(libro, nombre, columnas) {
  var hoja = libro.getSheetByName(nombre);
  if (!hoja) {
    hoja = libro.insertSheet(nombre);
  }
  if (hoja.getLastRow() === 0) {
    hoja.getRange(1, 1, 1, columnas.length).setValues([columnas]).setFontWeight('bold');
    hoja.setFrozenRows(1);
  }
  return hoja;
}

var MESES = ['ENERO', 'FEBRERO', 'MARZO', 'ABRIL', 'MAYO', 'JUNIO', 'JULIO',
             'AGOSTO', 'SEPTIEMBRE', 'OCTUBRE', 'NOVIEMBRE', 'DICIEMBRE'];

/**
 * Lleva el monto a la hoja del mes del cashflow: fila de la categoria,
 * columna del dia.
 *
 * No reemplaza lo que haya en la celda: lo SUMA, dejandolo como formula.
 * Si la celda decia =5000+7500 queda =5000+7500+11290. Es la misma forma en
 * que Luis la escribe a mano, asi que el historial de lo que compuso ese dia
 * no se pierde y el numero se puede desarmar mirando.
 *
 * Si algo no calza -no existe la hoja del mes, no aparece el dia, no esta la
 * categoria- no escribe nada y devuelve el motivo. Escribir en la celda
 * equivocada de un cashflow es peor que no escribir.
 */
function alCashflow(libro, datos, fecha) {
  if (!fecha || fecha.length < 10) return 'sin fecha, no se llevo al cashflow';
  var total = Number(datos.total);
  if (!total || total <= 0) return 'sin total, no se llevo al cashflow';

  var mes = Number(fecha.substring(5, 7));
  var dia = Number(fecha.substring(8, 10));
  var nombreMes = MESES[mes - 1];

  var hoja = hojaDelMes(libro, nombreMes);
  if (!hoja) return 'no existe la hoja GASTOS ' + nombreMes;

  var col = columnaDelDia(hoja, dia);
  if (col < 0) return 'no encontre el dia ' + dia + ' en GASTOS ' + nombreMes;

  var fila = filaDeCategoria(hoja, datos.categoria, Number(datos.categoriaFila));
  if (fila < 0) return 'no encontre la categoria ' + datos.categoria;

  var celda = hoja.getRange(fila, col);
  var formula = celda.getFormula();
  var valor = celda.getValue();

  if (formula && formula.charAt(0) === '=') {
    celda.setFormula(formula + '+' + total);
  } else if (valor !== '' && valor !== null && !isNaN(Number(valor))) {
    celda.setFormula('=' + Number(valor) + '+' + total);
  } else {
    celda.setValue(total);
  }
  return 'GASTOS ' + nombreMes + ' · ' + datos.categoria + ' · dia ' + dia;
}

/**
 * La hoja del mes. Se aceptan las dos formas en que Luis las nombra:
 * "GASTOS AGOSTO" y "GASTOS AGOSTO 2026".
 *
 * Si no existe, se copia la ultima hoja GASTOS que haya y se le borran los
 * dias. Copiar y no crear de cero es a proposito: asi la hoja nueva hereda
 * las categorias, los presupuestos y las formulas de totales que Luis
 * ajusto a mano, en vez de una version mia que se le va a parecer poco.
 */
function hojaDelMes(libro, nombreMes) {
  var exacto = libro.getSheetByName('GASTOS ' + nombreMes);
  if (exacto) return exacto;

  var hojas = libro.getSheets();
  var ultima = null;
  for (var i = 0; i < hojas.length; i++) {
    var n = hojas[i].getName();
    if (n.indexOf('GASTOS ' + nombreMes) === 0) return hojas[i];
    if (n.indexOf('GASTOS ') === 0) ultima = hojas[i];
  }
  if (!ultima) return null;

  var nueva = ultima.copyTo(libro).setName('GASTOS ' + nombreMes);
  // Se borran solo las celdas de dias; las categorias, los presupuestos y
  // las filas de totales quedan como estaban.
  nueva.getRange(4, 7, 23, 31).clearContent();
  return nueva;
}

/**
 * La columna del dia, contando desde donde esta el 1.
 *
 * No se busca el numero del dia directo porque la fila empieza con los
 * ultimos dias del mes anterior: en la hoja de agosto la fila arranca en 29,
 * 30, 31 y recien despues viene el 1. Buscar "29" encontraria el de julio.
 */
function columnaDelDia(hoja, dia) {
  var fila = hoja.getRange(3, 7, 1, 40).getValues()[0];
  var inicio = -1;
  for (var i = 0; i < fila.length; i++) {
    if (Number(fila[i]) === 1) { inicio = i; break; }
  }
  if (inicio < 0) return -1;
  var idx = inicio + (dia - 1);
  if (idx < 0 || idx >= fila.length) return -1;
  if (Number(fila[idx]) !== dia) return -1;
  return 7 + idx;
}

/**
 * La fila de la categoria. Manda el nombre, no el numero.
 *
 * El numero de fila que manda la app es solo la red de seguridad: si Luis
 * agrega o mueve una categoria en su planilla, el nombre sigue calzando y el
 * numero ya no. Buscar por nombre primero es lo que hace que la integracion
 * sobreviva a que el duenio edite su propia hoja.
 */
function filaDeCategoria(hoja, nombre, filaSugerida) {
  if (nombre) {
    var col = hoja.getRange(4, 4, 23, 1).getValues();
    var buscado = String(nombre).trim().toUpperCase();
    for (var i = 0; i < col.length; i++) {
      if (String(col[i][0]).trim().toUpperCase() === buscado) return 4 + i;
    }
  }
  if (filaSugerida >= 4 && filaSugerida <= 26) return filaSugerida;
  return -1;
}

/**
 * El Resumen son formulas, no datos copiados.
 *
 * Asi se recalcula solo cada vez que llega una boleta nueva. Si fueran
 * valores escritos por el script, habria que reescribirlos en cada envio y
 * cualquier envio fallido dejaria el resumen mintiendo.
 */
function crearResumen(libro) {
  var hoja = libro.getSheetByName(HOJA_RES);
  if (hoja) return hoja;
  hoja = libro.insertSheet(HOJA_RES);

  hoja.getRange('A1').setValue('Por dia').setFontWeight('bold');
  hoja.getRange('A2').setFormula(
    "=IFERROR(QUERY(Documentos!A:J, \"select B, count(A), sum(J) " +
    "where B is not null and B <> 'Fecha' group by B order by B desc " +
    "label count(A) 'Documentos', sum(J) 'Total'\", 1), \"\")");

  hoja.getRange('E1').setValue('Por mes').setFontWeight('bold');
  hoja.getRange('E2').setFormula(
    "=IFERROR(QUERY(Documentos!A:J, \"select C, count(A), sum(J) " +
    "where C is not null and C <> 'Mes' group by C order by C desc " +
    "label count(A) 'Documentos', sum(J) 'Total'\", 1), \"\")");

  hoja.getRange('I1').setValue('Por ano').setFontWeight('bold');
  hoja.getRange('I2').setFormula(
    "=IFERROR(QUERY(Documentos!A:J, \"select D, count(A), sum(J) " +
    "where D is not null and D <> 'Ano' group by D order by D desc " +
    "label count(A) 'Documentos', sum(J) 'Total'\", 1), \"\")");

  hoja.getRange('M1').setValue('Lo que mas gastas').setFontWeight('bold');
  hoja.getRange('M2').setFormula(
    "=IFERROR(QUERY(Detalle!A:G, \"select D, sum(G) " +
    "where D is not null and D <> 'Descripcion' group by D " +
    "order by sum(G) desc limit 25 label sum(G) 'Total'\", 1), \"\")");

  return hoja;
}
